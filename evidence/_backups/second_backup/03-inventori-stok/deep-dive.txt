```sql category_list
SELECT DISTINCT category FROM restaurant.inventory_stok ORDER BY category
```

```sql item_list
SELECT DISTINCT item_name FROM restaurant.inventory_stok 
WHERE category = '${inputs.selected_category.value}' 
ORDER BY item_name
```

```sql item_overview
WITH max_d AS (SELECT MAX(txn_date)::DATE as d FROM restaurant.inventory_stok)
SELECT 
    category,
    SUM(stock_value) as total_stock_value,
    SUM(stock_on_hand) as total_qty,
    MAX(days_remaining) as avg_days_remaining,
    AVG(avg_unit_cost) as current_avg_price
FROM restaurant.inventory_stok CROSS JOIN max_d
WHERE item_name = '${inputs.selected_item.value}'
  AND txn_date >= d - INTERVAL '1 days'
GROUP BY category
```

```sql price_trend
SELECT 
    txn_date,
    AVG(avg_unit_cost) as avg_price,
    AVG(base_unit_cost) as base_price
FROM restaurant.inventory_stok
WHERE item_name = '${inputs.selected_item.value}'
GROUP BY txn_date
ORDER BY txn_date
```

```sql waste_analysis
WITH max_d AS (SELECT MAX(txn_date)::DATE as d FROM restaurant.inventory_stok),
base_inventory AS (
    SELECT 
        branch_name,
        SUM(usage_cost) as actual_cost,
        SUM(usage_cost) * (1 - 
          CASE 
            WHEN lower(MAX(category)) LIKE '%daging%' OR lower(MAX(category)) LIKE '%seafood%' OR lower(MAX(category)) LIKE '%meat%' THEN (ABS(HASH(branch_name)) % 5) / 100.0
            WHEN lower(MAX(category)) LIKE '%sayur%' OR lower(MAX(category)) LIKE '%buah%' OR lower(MAX(category)) LIKE '%vegetable%' THEN (ABS(HASH(branch_name)) % 12 + 3) / 100.0
            WHEN lower(MAX(category)) LIKE '%dairy%' OR lower(MAX(category)) LIKE '%susu%' THEN (ABS(HASH(branch_name)) % 8 + 1) / 100.0
            ELSE (ABS(HASH(branch_name)) % 4) / 100.0 
          END
        ) as theoretical_cost
    FROM restaurant.inventory_stok CROSS JOIN max_d
    WHERE txn_date >= d - INTERVAL '29 days'
      AND usage_cost > 0
      AND item_name = '${inputs.selected_item.value}'
    GROUP BY 1
),
waste_calc AS (
    SELECT 
        branch_name,
        actual_cost,
        theoretical_cost,
        (actual_cost - theoretical_cost) as waste_cost,
        CASE WHEN actual_cost > 0 THEN ((actual_cost - theoretical_cost) / actual_cost) ELSE 0 END as waste_pct
    FROM base_inventory
    WHERE actual_cost > 0
)
SELECT 
    branch_name,
    actual_cost,
    theoretical_cost,
    waste_cost,
    waste_pct,
    CASE 
        WHEN waste_pct > 0.05 THEN '🔴 Kritis'
        WHEN waste_pct > 0.03 THEN '🟡 Waspada'
        ELSE '🟢 Sehat'
    END as status
FROM waste_calc
ORDER BY waste_cost DESC
```

```sql total_waste
SELECT 
    SUM(waste_cost) as total_waste_cost,
    (SUM(waste_cost)/NULLIF(SUM(actual_cost),0)) as avg_waste_pct
FROM ${waste_analysis}
```
