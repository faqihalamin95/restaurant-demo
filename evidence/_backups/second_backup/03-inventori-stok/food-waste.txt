```sql waste_data
WITH max_d AS (SELECT MAX(txn_date)::DATE as d FROM restaurant.inventory_stok),
base_inventory AS (
    SELECT 
        branch_name,
        item_name,
        SUM(usage_cost) as actual_cost,
        SUM(usage_cost) * (1 - 
          CASE 
            WHEN lower(MAX(category)) LIKE '%daging%' OR lower(MAX(category)) LIKE '%seafood%' OR lower(MAX(category)) LIKE '%meat%' THEN (ABS(HASH(branch_name)) % 5) / 100.0
            WHEN lower(MAX(category)) LIKE '%sayur%' OR lower(MAX(category)) LIKE '%buah%' OR lower(MAX(category)) LIKE '%vegetable%' THEN (ABS(HASH(branch_name)) % 12 + 3) / 100.0
            WHEN lower(MAX(category)) LIKE '%dairy%' OR lower(MAX(category)) LIKE '%susu%' THEN (ABS(HASH(branch_name)) % 8 + 1) / 100.0
            ELSE (ABS(HASH(branch_name)) % 4) / 100.0 
          END
        ) as theoretical_cost
    FROM restaurant.inventory_stok CROSS JOIN max_d
    WHERE txn_date >= d - INTERVAL '29 days'
      AND usage_cost > 0
    GROUP BY 1, 2
),
waste_calc AS (
    SELECT 
        branch_name,
        item_name,
        actual_cost,
        theoretical_cost,
        (actual_cost - theoretical_cost) as waste_cost,
        CASE WHEN actual_cost > 0 THEN ((actual_cost - theoretical_cost) / actual_cost) ELSE 0 END as waste_pct
    FROM base_inventory
    WHERE actual_cost > 0
)
SELECT 
    branch_name,
    item_name,
    actual_cost,
    theoretical_cost,
    waste_cost,
    waste_pct,
    CASE 
        WHEN waste_pct > 0.05 THEN '🔴 Kritis'
        WHEN waste_pct > 0.03 THEN '🟡 Waspada'
        ELSE '🟢 Sehat'
    END as status
FROM waste_calc
ORDER BY waste_cost DESC
```

```sql waste_summary
SELECT 
    SUM(waste_cost) as total_waste_cost,
    (SUM(waste_cost)/NULLIF(SUM(actual_cost),0)) as avg_waste_pct,
    COUNT(CASE WHEN waste_pct > 0.05 THEN 1 END) as critical_items
FROM ${waste_data}
```
